/*
 * ============================================================
 *  MiniOS-Sim: Simulador de Núcleo de Sistema Operativo
 * ============================================================
 *  Módulo: Gestión de Procesos (CPU)
 *
 *  Implementa:
 *    - PCB (Bloque de Control de Procesos)
 *    - Cola de procesos listos
 *    - Algoritmos: FCFS, SJF (no apropiativo), SRTF (apropiativo),
 *      Round Robin (quantum configurable), Prioridades (con y sin aging)
 *    - Métricas: Turnaround, Waiting, Response Time, Throughput
 *
 *  Patrón de diseño usado: STRATEGY
 *    -> Cada algoritmo de planificación es una clase que hereda
 *       de Scheduler y define su propia forma de planificar.
 *       Esto permite añadir nuevos algoritmos sin tocar el resto
 *       del programa (principio Open/Closed).
 * ============================================================
 */

#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
#include <iomanip>
#include <limits>
#include <memory>
#include <string>
#include <sstream>

using namespace std;

// ------------------------------------------------------------
//  PCB: Bloque de Control de Procesos
// ------------------------------------------------------------
struct PCB {
    int pid;
    int arrivalTime;
    int burstTime;      // tiempo de ráfaga original (CPU)
    int priority;        // menor número = mayor prioridad
    int remainingTime;   // usado por algoritmos apropiativos

    // Campos que se calculan durante la simulación
    int startTime      = -1; // primera vez que entra a CPU (para response time)
    int completionTime = -1;
    int waitingTime     = 0;
    int turnaroundTime  = 0;
    int responseTime    = -1;

    // Para aging (envejecimiento de prioridad)
    int originalPriority = 0;
    int waitCounter       = 0; // ciclos esperando en la cola listos

    PCB(int id, int arrival, int burst, int prio)
        : pid(id), arrivalTime(arrival), burstTime(burst), priority(prio),
          remainingTime(burst), originalPriority(prio) {}
};

// Un "evento" de ejecución para armar el diagrama de Gantt en texto
struct GanttBlock {
    int pid;
    int start;
    int end;
};

// ------------------------------------------------------------
//  Resultado de una simulación (lo que necesita Metrics)
// ------------------------------------------------------------
struct SimulationResult {
    string algorithmName;
    vector<PCB> processes;       // con los tiempos ya calculados
    vector<GanttBlock> gantt;
    int totalTime = 0;           // tiempo total de la simulación (para throughput)
};

// ------------------------------------------------------------
//  Clase base abstracta (Strategy) para los algoritmos
// ------------------------------------------------------------
class Scheduler {
public:
    virtual ~Scheduler() = default;
    virtual SimulationResult run(vector<PCB> processes) = 0; // recibe copia
    virtual string name() const = 0;
};

// ------------------------------------------------------------
//  1) FCFS - First Come First Served
// ------------------------------------------------------------
class FCFSScheduler : public Scheduler {
public:
    string name() const override { return "FCFS (First-Come, First-Served)"; }

    SimulationResult run(vector<PCB> processes) override {
        SimulationResult result;
        result.algorithmName = name();

        sort(processes.begin(), processes.end(), [](const PCB& a, const PCB& b) {
            return a.arrivalTime < b.arrivalTime;
        });

        int currentTime = 0;
        for (auto& p : processes) {
            if (currentTime < p.arrivalTime) currentTime = p.arrivalTime;

            p.startTime = currentTime;
            p.responseTime = p.startTime - p.arrivalTime;

            int end = currentTime + p.burstTime;
            result.gantt.push_back({p.pid, currentTime, end});

            currentTime = end;
            p.completionTime = currentTime;
            p.turnaroundTime = p.completionTime - p.arrivalTime;
            p.waitingTime = p.turnaroundTime - p.burstTime;
        }

        result.totalTime = currentTime;
        result.processes = processes;
        return result;
    }
};

// ------------------------------------------------------------
//  2) SJF - Shortest Job First (No apropiativo)
// ------------------------------------------------------------
class SJFScheduler : public Scheduler {
public:
    string name() const override { return "SJF (Shortest Job First - No apropiativo)"; }

    SimulationResult run(vector<PCB> processes) override {
        SimulationResult result;
        result.algorithmName = name();

        int n = processes.size();
        vector<bool> done(n, false);
        int completed = 0;
        int currentTime = 0;

        while (completed < n) {
            int idx = -1;
            int best = numeric_limits<int>::max();

            for (int i = 0; i < n; i++) {
                if (!done[i] && processes[i].arrivalTime <= currentTime &&
                    processes[i].burstTime < best) {
                    best = processes[i].burstTime;
                    idx = i;
                }
            }

            if (idx == -1) {
                // No hay proceso listo: la CPU está inactiva, avanzamos el tiempo
                currentTime++;
                continue;
            }

            PCB& p = processes[idx];
            p.startTime = currentTime;
            p.responseTime = p.startTime - p.arrivalTime;

            int end = currentTime + p.burstTime;
            result.gantt.push_back({p.pid, currentTime, end});

            currentTime = end;
            p.completionTime = currentTime;
            p.turnaroundTime = p.completionTime - p.arrivalTime;
            p.waitingTime = p.turnaroundTime - p.burstTime;

            done[idx] = true;
            completed++;
        }

        result.totalTime = currentTime;
        result.processes = processes;
        return result;
    }
};

// ------------------------------------------------------------
//  3) SRTF - Shortest Remaining Time First (Apropiativo)
// ------------------------------------------------------------
class SRTFScheduler : public Scheduler {
public:
    string name() const override { return "SRTF (Shortest Remaining Time First - Apropiativo)"; }

    SimulationResult run(vector<PCB> processes) override {
        SimulationResult result;
        result.algorithmName = name();

        int n = processes.size();
        int completed = 0;
        int currentTime = 0;
        int lastPid = -1; // para agrupar bloques consecutivos del mismo proceso en el Gantt

        while (completed < n) {
            int idx = -1;
            int best = numeric_limits<int>::max();

            for (int i = 0; i < n; i++) {
                if (processes[i].arrivalTime <= currentTime &&
                    processes[i].remainingTime > 0 &&
                    processes[i].remainingTime < best) {
                    best = processes[i].remainingTime;
                    idx = i;
                }
            }

            if (idx == -1) {
                currentTime++;
                lastPid = -1;
                continue;
            }

            PCB& p = processes[idx];
            if (p.startTime == -1) {
                p.startTime = currentTime;
                p.responseTime = p.startTime - p.arrivalTime;
            }

            // Ejecuta 1 unidad de tiempo (simulación por quantum de 1 para permitir apropiación)
            if (lastPid == p.pid && !result.gantt.empty()) {
                result.gantt.back().end++;
            } else {
                result.gantt.push_back({p.pid, currentTime, currentTime + 1});
            }
            lastPid = p.pid;

            p.remainingTime--;
            currentTime++;

            if (p.remainingTime == 0) {
                p.completionTime = currentTime;
                p.turnaroundTime = p.completionTime - p.arrivalTime;
                p.waitingTime = p.turnaroundTime - p.burstTime;
                completed++;
                lastPid = -1;
            }
        }

        result.totalTime = currentTime;
        result.processes = processes;
        return result;
    }
};

// ------------------------------------------------------------
//  4) Round Robin (quantum configurable)
// ------------------------------------------------------------
class RoundRobinScheduler : public Scheduler {
    int quantum;
public:
    explicit RoundRobinScheduler(int q) : quantum(q) {}
    string name() const override {
        ostringstream oss; oss << "Round Robin (quantum = " << quantum << ")"; return oss.str();
    }

    SimulationResult run(vector<PCB> processes) override {
        SimulationResult result;
        result.algorithmName = name();

        int n = processes.size();
        sort(processes.begin(), processes.end(), [](const PCB& a, const PCB& b) {
            return a.arrivalTime < b.arrivalTime;
        });

        queue<int> readyQueue;
        vector<bool> inQueue(n, false);
        int currentTime = 0;
        int completed = 0;

        // Mete el primero que llega
        int nextArrivalIdx = 0;
        auto enqueueArrivals = [&](int uptoTime) {
            while (nextArrivalIdx < n && processes[nextArrivalIdx].arrivalTime <= uptoTime) {
                readyQueue.push(nextArrivalIdx);
                inQueue[nextArrivalIdx] = true;
                nextArrivalIdx++;
            }
        };

        currentTime = processes[0].arrivalTime;
        enqueueArrivals(currentTime);

        while (completed < n) {
            if (readyQueue.empty()) {
                currentTime++;
                enqueueArrivals(currentTime);
                continue;
            }

            int idx = readyQueue.front();
            readyQueue.pop();
            PCB& p = processes[idx];

            if (p.startTime == -1) {
                p.startTime = currentTime;
                p.responseTime = p.startTime - p.arrivalTime;
            }

            int execTime = min(quantum, p.remainingTime);
            result.gantt.push_back({p.pid, currentTime, currentTime + execTime});

            currentTime += execTime;
            p.remainingTime -= execTime;

            // Los que llegaron durante esta ejecución entran ANTES que el proceso actual
            enqueueArrivals(currentTime);

            if (p.remainingTime > 0) {
                readyQueue.push(idx);
            } else {
                p.completionTime = currentTime;
                p.turnaroundTime = p.completionTime - p.arrivalTime;
                p.waitingTime = p.turnaroundTime - p.burstTime;
                completed++;
            }
        }

        result.totalTime = currentTime;
        result.processes = processes;
        return result;
    }
};

// ------------------------------------------------------------
//  5) Prioridades (con o sin aging), No apropiativo
// ------------------------------------------------------------
class PriorityScheduler : public Scheduler {
    bool useAging;
    int agingInterval; // cada cuántas unidades de espera sube la prioridad
public:
    explicit PriorityScheduler(bool aging, int interval = 5)
        : useAging(aging), agingInterval(interval) {}

    string name() const override {
        return useAging ? "Prioridades (con aging)" : "Prioridades (sin aging)";
    }

    SimulationResult run(vector<PCB> processes) override {
        SimulationResult result;
        result.algorithmName = name();

        int n = processes.size();
        vector<bool> done(n, false);
        int completed = 0;
        int currentTime = 0;

        while (completed < n) {
            int idx = -1;
            int bestPriority = numeric_limits<int>::max();

            for (int i = 0; i < n; i++) {
                if (!done[i] && processes[i].arrivalTime <= currentTime) {
                    if (processes[i].priority < bestPriority) {
                        bestPriority = processes[i].priority;
                        idx = i;
                    }
                }
            }

            if (idx == -1) {
                currentTime++;
                // Aging: los procesos que ya llegaron pero siguen esperando ganan prioridad
                if (useAging) {
                    for (int i = 0; i < n; i++) {
                        if (!done[i] && processes[i].arrivalTime <= currentTime) {
                            processes[i].waitCounter++;
                            if (processes[i].waitCounter % agingInterval == 0 &&
                                processes[i].priority > 0) {
                                processes[i].priority--;
                            }
                        }
                    }
                }
                continue;
            }

            PCB& p = processes[idx];
            p.startTime = currentTime;
            p.responseTime = p.startTime - p.arrivalTime;

            int end = currentTime + p.burstTime;
            result.gantt.push_back({p.pid, currentTime, end});

            // Aging durante la ejecución de este proceso para los que esperan
            if (useAging) {
                for (int i = 0; i < n; i++) {
                    if (i != idx && !done[i] && processes[i].arrivalTime <= end) {
                        int waited = end - max(processes[i].arrivalTime, currentTime);
                        processes[i].waitCounter += waited;
                        int drops = processes[i].waitCounter / agingInterval;
                        processes[i].priority = max(0, processes[i].originalPriority - drops);
                    }
                }
            }

            currentTime = end;
            p.completionTime = currentTime;
            p.turnaroundTime = p.completionTime - p.arrivalTime;
            p.waitingTime = p.turnaroundTime - p.burstTime;

            done[idx] = true;
            completed++;
        }

        result.totalTime = currentTime;
        result.processes = processes;
        return result;
    }
};

// ------------------------------------------------------------
//  Metrics: cálculo y presentación de resultados
// ------------------------------------------------------------
class Metrics {
public:
    static void print(const SimulationResult& r) {
        cout << "\nAlgoritmo: " << r.algorithmName << "\n\n";

        // Ordena por PID solo para mostrar bonito
        vector<PCB> ordered = r.processes;
        sort(ordered.begin(), ordered.end(), [](const PCB& a, const PCB& b) {
            return a.pid < b.pid;
        });

        cout << left
             << setw(6)  << "PID"
             << setw(10) << "Llegada"
             << setw(10) << "Rafaga"
             << setw(12) << "Finaliza"
             << setw(12) << "Retorno"
             << setw(10) << "Espera"
             << setw(12) << "Respuesta" << "\n";

        double sumTurnaround = 0, sumWaiting = 0, sumResponse = 0;
        for (auto& p : ordered) {
            cout << left
                 << setw(6)  << p.pid
                 << setw(10) << p.arrivalTime
                 << setw(10) << p.burstTime
                 << setw(12) << p.completionTime
                 << setw(12) << p.turnaroundTime
                 << setw(10) << p.waitingTime
                 << setw(12) << p.responseTime << "\n";

            sumTurnaround += p.turnaroundTime;
            sumWaiting    += p.waitingTime;
            sumResponse   += p.responseTime;
        }

        int n = ordered.size();
        double throughput = (double)n / r.totalTime;

        cout << "\n";
        cout << fixed << setprecision(2);
        cout << "Tiempo de Retorno promedio   : " << (sumTurnaround / n) << "\n";
        cout << "Tiempo de Espera promedio    : " << (sumWaiting / n) << "\n";
        cout << "Tiempo de Respuesta promedio : " << (sumResponse / n) << "\n";
        cout << "Throughput                   : " << throughput << " procesos/unidad de tiempo\n";
        cout << "Tiempo total de simulacion   : " << r.totalTime << "\n";
    }

    static void printGantt(const SimulationResult& r) {
        cout << "\nDiagrama de Gantt (" << r.algorithmName << "):\n";
        for (size_t i = 0; i < r.gantt.size(); i++) {
            auto& b = r.gantt[i];
            cout << "P" << b.pid << " (" << b.start << "-" << b.end << ")";
            if (i + 1 < r.gantt.size()) cout << "  ->  ";
        }
        cout << "\n";
    }
};

//  Utilidades de entrada de datos

vector<PCB> loadSampleProcesses() {
    // Set de procesos de ejemplo para probar rápido sin digitar nada
    vector<PCB> processes;
    processes.push_back(PCB(1, 0, 8, 2));
    processes.push_back(PCB(2, 1, 4, 1));
    processes.push_back(PCB(3, 2, 9, 3));
    processes.push_back(PCB(4, 3, 5, 0));
    processes.push_back(PCB(5, 4, 2, 4));
    return processes;
}

vector<PCB> readProcessesFromUser() {
    vector<PCB> processes;
    int n;
    cout << "Cuantos procesos desea ingresar? ";
    cin >> n;

    for (int i = 0; i < n; i++) {
        int arrival, burst, priority;
        cout << "\n--- Proceso P" << (i + 1) << " ---\n";
        cout << "Tiempo de llegada: ";
        cin >> arrival;
        cout << "Tiempo de rafaga (burst): ";
        cin >> burst;
        cout << "Prioridad (0 = mas alta): ";
        cin >> priority;
        processes.push_back(PCB(i + 1, arrival, burst, priority));
    }
    return processes;
}

int main() {
    cout << "MiniOS-Sim: Simulador de Nucleo de Sistema Operativo\n";
    cout << "Modulo: Gestion de Procesos (CPU)\n";

    vector<PCB> baseProcesses;

    cout << "\nDesea usar procesos de ejemplo (1) o ingresar los suyos (2)? ";
    int opt;
    cin >> opt;

    if (opt == 2) {
        baseProcesses = readProcessesFromUser();
    } else {
        baseProcesses = loadSampleProcesses();
        cout << "\nUsando 5 procesos de ejemplo (P1..P5).\n";
    }

    bool salir = false;
    while (!salir) {
        cout << "\nSeleccione el algoritmo de planificacion:\n";
        cout << "  1) FCFS\n";
        cout << "  2) SJF (no apropiativo)\n";
        cout << "  3) SRTF (apropiativo)\n";
        cout << "  4) Round Robin\n";
        cout << "  5) Prioridades (sin aging)\n";
        cout << "  6) Prioridades (con aging)\n";
        cout << "  7) Ejecutar TODOS y comparar\n";
        cout << "  0) Salir\n";
        cout << "Opcion: ";

        int choice;
        cin >> choice;

        unique_ptr<Scheduler> scheduler;
        int quantum;

        switch (choice) {
            case 1:
                scheduler = unique_ptr<Scheduler>(new FCFSScheduler());
                break;
            case 2:
                scheduler = unique_ptr<Scheduler>(new SJFScheduler());
                break;
            case 3:
                scheduler = unique_ptr<Scheduler>(new SRTFScheduler());
                break;
            case 4:
                cout << "Ingrese el quantum: ";
                cin >> quantum;
                scheduler = unique_ptr<Scheduler>(new RoundRobinScheduler(quantum));
                break;
            case 5:
                scheduler = unique_ptr<Scheduler>(new PriorityScheduler(false));
                break;
            case 6:
                scheduler = unique_ptr<Scheduler>(new PriorityScheduler(true));
                break;
            case 7: {
                // Ejecuta todos con los mismos procesos base y compara
                vector<unique_ptr<Scheduler>> all;
                all.push_back(unique_ptr<Scheduler>(new FCFSScheduler()));
                all.push_back(unique_ptr<Scheduler>(new SJFScheduler()));
                all.push_back(unique_ptr<Scheduler>(new SRTFScheduler()));
                all.push_back(unique_ptr<Scheduler>(new RoundRobinScheduler(4)));
                all.push_back(unique_ptr<Scheduler>(new PriorityScheduler(false)));
                all.push_back(unique_ptr<Scheduler>(new PriorityScheduler(true)));

                for (auto& s : all) {
                    SimulationResult r = s->run(baseProcesses);
                    Metrics::print(r);
                    Metrics::printGantt(r);
                }
                continue; // vuelve al menu sin pasar por el bloque de abajo
            }
            case 0:
                salir = true;
                continue;
            default:
                cout << "Opcion invalida.\n";
                continue;
        }

        SimulationResult result = scheduler->run(baseProcesses);
        Metrics::print(result);
        Metrics::printGantt(result);
    }

    cout << "\nSaliendo del simulador. Hasta la proxima!\n";
    return 0;
}
